The milestone's done check

The done-when, run as written
Setup

Save this as /tmp/check-m3.js (it is not in the repo). It reads the groups, then presses the first question's primary answer with every POST refused in the page, so nothing is really answered:

(async () => {
  const q = (s, r = document) => [...r.querySelectorAll(s)];
  const groups = q(".cc-project > section.rack").map((r) => r.querySelector("h4 > span:not(.n)")?.textContent.trim());
  const rows = q(".rack.needs article[data-question]").map((a) => ({
    mission: a.dataset.mission, question: a.dataset.question,
    answers: q(".inbox-answers button", a).map((b) => b.textContent),
  }));
  const posted = [];
  const real = window.fetch;
  window.fetch = (url, opts = {}) => {
    if ((opts.method ?? "GET") === "POST") { posted.push(`POST ${url} ${opts.body ?? ""}`); return Promise.resolve(new Response('{"error":"refused by the check"}', { status: 409 })); }
    return real(url, opts);
  };
  const first = document.querySelector(".rack.needs article[data-question]");
  first?.querySelector(".inbox-answers button.primary")?.click();
  await new Promise((r) => setTimeout(r, 600));
  window.fetch = real;
  const hint = first?.querySelector("[role=status]")?.textContent ?? null;
  return JSON.stringify({ groups, rows, posted, hint }, null, 1);
})()
Steps
  1. Best while the mission is paused (between q1 and q2). In the integration worktree: node tools/cdp.mjs "http://localhost:8767/?project=maverick&view=workspace" --eval /tmp/check-m3.js --wait 6000
  2. npm run lint
Expected
  • groups starts with Needs you, Working, Ready to review, Done in that order (any empty one missing, the order kept), then Brainstorms.
  • While paused: rows holds {"mission": "mission-page-and-inbox", "question": "paused", "answers": ["resume"]}, posted is ["POST /api/missions/mission-page-and-inbox/resume?project=maverick {}"], and hint is resume failed: refused by the check. (Not paused: rows, posted empty, hint null.)
  • Lint exits 0. (It is node --check only: a syntax pass.)
